async function saveMatchesToCloud() { if(!currentTournamentId) { alert("Tournament ID missing. Ensure Step 1 completed successfully."); return; } if(finalLockedMatches.length === 0) { alert("No matches found to deploy."); return; } const btn = document.getElementById('btnDeploy'); btn.innerText = "⏳ Injecting Fixtures & Squads to Supabase..."; let generatedOutput = []; for (let m of finalLockedMatches) { let pin = m.pin || generateUniquePin(); let team1UUID = activeTeamMap[m.t1] || null; let team2UUID = activeTeamMap[m.t2] || null; let tName = document.getElementById('t-name').value; if(team1UUID && tournamentSquadsCache[m.t1]) { let s1 = tournamentSquadsCache[m.t1].map(pid => ({ tournament_id: currentTournamentId, team_id: team1UUID, player_id: pid })); await db.from('tournament_squads').upsert(s1, {onConflict: 'tournament_id, team_id, player_id'}); } if(team2UUID && tournamentSquadsCache[m.t2]) { let s2 = tournamentSquadsCache[m.t2].map(pid => ({ tournament_id: currentTournamentId, team_id: team2UUID, player_id: pid })); await db.from('tournament_squads').upsert(s2, {onConflict: 'tournament_id, team_id, player_id'}); } let matchPayload = { match_id: m.id, tournament_id: currentTournamentId, team_a_id: team1UUID, team_b_id: team2UUID, scorer_pin: pin, // NEW: We officially flag this match as "upcoming" in the cloud full_state: { tournament: tName, team1: m.t1, team2: m.t2, matchName: m.name, match_status: 'upcoming' } }; await db.from('matches').upsert([matchPayload]); generatedOutput.push({ id: m.id, name: m.name, t1: m.t1, t2: m.t2, pin: pin }); } btn.style.display = 'none'; displayPins(generatedOutput); }
